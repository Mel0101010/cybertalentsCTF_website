Mission #14: Analyse de Protocoles Réseau

STATUT: EN COURS

📋 Briefing de Mission

Dans cette mission, vous devez créer un programme qui analyse le trafic réseau pour identifier et analyser différents protocoles. Le serveur vous enverra des paquets réseau à analyser pour déterminer leur protocole, leur contenu et leur comportement.

Votre programme devra extraire les informations importantes, détecter les anomalies et générer un rapport d'analyse.

📡 Informations du Serveur

IP : ?.?.?.?

Port : 9014

Protocole : TCP

📝 Protocoles à Analyser

Les paquets peuvent appartenir à différents protocoles :

  • HTTP : Protocole de transfert hypertexte
  • DNS : Système de noms de domaine
  • FTP : Protocole de transfert de fichiers
  • SMTP : Protocole de transfert de courrier
  • SSH : Protocole de connexion sécurisée

import socket
import re
from datetime import datetime

class ProtocolAnalyzer:
    def __init__(self):
        self.protocol_patterns = {
            'http': [
                (r'^(GET|POST|PUT|DELETE|HEAD|OPTIONS|PATCH) .* HTTP/\d\.\d$', 'request'),
                (r'^HTTP/\d\.\d \d{3} .*$', 'response'),
                (r'^Host: .*$', 'header'),
                (r'^User-Agent: .*$', 'header'),
                (r'^Content-Type: .*$', 'header')
            ],
            'dns': [
                (r'Transaction ID: 0x[0-9a-f]{4}$', 'header'),
                (r'Flags: .*$', 'header'),
                (r'Questions: \d+$', 'header'),
                (r'Answer RRs: \d+$', 'header'),
                (r'.* IN (A|AAAA|CNAME|MX|TXT)$', 'query')
            ],
            'ftp': [
                (r'^(USER|PASS|ACCT|CWD|CDUP|SMNT|QUIT|REIN|PORT|PASV|TYPE|STRU|MODE|RETR|STOR|STOU|APPE|ALLO|REST|RNFR|RNTO|ABOR|DELE|RMD|MKD|PWD|LIST|NLST|SITE|SYST|STAT|HELP|NOOP) .*$', 'command'),
                (r'^\d{3} .*$', 'response')
            ],
            'smtp': [
                (r'^(HELO|EHLO|MAIL|RCPT|DATA|RSET|VRFY|EXPN|HELP|NOOP|QUIT) .*$', 'command'),
                (r'^\d{3} .*$', 'response'),
                (r'^From: .*$', 'header'),
                (r'^To: .*$', 'header'),
                (r'^Subject: .*$', 'header')
            ],
            'ssh': [
                (r'^SSH-\d\.\d-.*$', 'version'),
                (r'^.* key exchange .*$', 'handshake'),
                (r'^.* encryption .*$', 'handshake'),
                (r'^.* authentication .*$', 'handshake')

        self.anomaly_patterns = {
            'http': [

                (r'User-Agent: (sqlmap|nikto|nmap|metasploit)', 'suspicious_agent'),
                (r'Content-Type: (application/x-php|application/x-asp|application/x-jsp)', 'suspicious_content')
            ],
            'dns': [
                (r'.*\.(malware|virus|trojan|backdoor)\.com$', 'suspicious_domain'),
                (r'TXT ".*[<>].*"', 'suspicious_txt'),
                (r'CNAME .*\.(tor|vpn|proxy)\.', 'suspicious_cname')
            ],
            'ftp': [
                (r'USER (admin|root|administrator)', 'suspicious_user'),

                (r'RETR .*\.(conf|passwd|shadow)$', 'suspicious_download')
            ],
            'smtp': [

    def extract_packet_info(self, packet_data):
        """Extrait les informations de base du paquet."""
        info = {}
        
        # Extraire les informations de base
        source_match = re.search(r'Source: (.*?):(\d+)', packet_data)
        dest_match = re.search(r'Destination: (.*?):(\d+)', packet_data)
        protocol_match = re.search(r'Protocol: (.*)', packet_data)
        length_match = re.search(r'Length: (\d+) bytes', packet_data)
        
        if all([source_match, dest_match, protocol_match, length_match]):
            info['source'] = {
                'ip': source_match.group(1),
                'port': int(source_match.group(2))
            }
            info['destination'] = {
                'ip': dest_match.group(1),
                'port': int(dest_match.group(2))
            }
            info['protocol'] = protocol_match.group(1)
            info['length'] = int(length_match.group(1))
        
        # Extraire les flags TCP si présents
        flags_match = re.search(r'Flags: (.*)', packet_data)
        if flags_match:
            info['flags'] = flags_match.group(1).split(',')
        
        # Extraire le checksum
        checksum_match = re.search(r'Checksum: (0x[0-9a-f]{4})', packet_data)
        if checksum_match:
            info['checksum'] = checksum_match.group(1)
        
        return info

    def extract_headers(self, packet_data):
        """Extrait les en-têtes du paquet."""
        headers_section = re.search(r'\[HEADERS\](.*?)(?=\[|$)', packet_data, re.DOTALL)
        if not headers_section:
            return []
        
        headers = []
        for line in headers_section.group(1).split('\n'):
            line = line.strip()
            if line:
                headers.append(line)
        
        return headers

    def extract_payload(self, packet_data):
        """Extrait le payload du paquet."""
        payload_section = re.search(r'\[PAYLOAD\](.*?)(?=\[|$)', packet_data, re.DOTALL)
        if not payload_section:
            return None
        
        return payload_section.group(1).strip()

    def identify_protocol(self, headers, payload):
        """Identifie le protocole basé sur les en-têtes et le payload."""
        protocol_scores = {
            'http': 0,
            'dns': 0,
            'ftp': 0,
            'smtp': 0,
            'ssh': 0
        }
        
        # Analyser les en-têtes
        for header in headers:
            for protocol, patterns in self.protocol_patterns.items():
                for pattern, _ in patterns:
                    if re.search(pattern, header, re.IGNORECASE):
                        protocol_scores[protocol] += 1
        
        # Analyser le payload si présent
        if payload:
            for protocol, patterns in self.protocol_patterns.items():
                for pattern, _ in patterns:
                    if re.search(pattern, payload, re.IGNORECASE):
                        protocol_scores[protocol] += 1
        
        # Déterminer le protocole avec le score le plus élevé
        max_score = max(protocol_scores.values())
        if max_score == 0:
            return 'Unknown'
        
        return max(protocol_scores.items(), key=lambda x: x[1])[0]

    def analyze_anomalies(self, protocol, headers, payload):
        """Analyse les anomalies dans le paquet."""
        anomalies = []
        
        if protocol in self.anomaly_patterns:
            # Vérifier les en-têtes
            for header in headers:
                for pattern, anomaly_type in self.anomaly_patterns[protocol]:
                    if callable(anomaly_type):
                        match = re.search(pattern, header, re.IGNORECASE)
                        if match:
                            result = anomaly_type(match)
                            if result:
                                anomalies.append(result)
                    elif re.search(pattern, header, re.IGNORECASE):
                        anomalies.append(anomaly_type)
            
            # Vérifier le payload si présent
            if payload:
                for pattern, anomaly_type in self.anomaly_patterns[protocol]:
                    if callable(anomaly_type):
                        match = re.search(pattern, payload, re.IGNORECASE)
                        if match:
                            result = anomaly_type(match)
                            if result:
                                anomalies.append(result)
                    elif re.search(pattern, payload, re.IGNORECASE):
                        anomalies.append(anomaly_type)
        
        return anomalies

    def analyze_packet(self, packet_data):
        """Analyse un paquet réseau et génère un rapport."""
        # Extraire les informations de base
        packet_info = self.extract_packet_info(packet_data)
        if not packet_info:
            return None
        
        # Extraire les en-têtes et le payload
        headers = self.extract_headers(packet_data)
        payload = self.extract_payload(packet_data)
        
        # Identifier le protocole
        protocol = self.identify_protocol(headers, payload)
        
        # Analyser les anomalies
        anomalies = self.analyze_anomalies(protocol, headers, payload)
        
        # Générer les détails spécifiques au protocole
        details = self.generate_protocol_details(protocol, headers, payload)
        
        # Déterminer le niveau de confiance
        confidence = 'HIGH' if len(details) >= 3 and not anomalies else 'MEDIUM' if len(details) >= 2 else 'LOW'
        
        return {
            'type': protocol.upper(),
            'info': packet_info,
            'details': details,
            'anomalies': anomalies,
            'confidence': confidence
        }
