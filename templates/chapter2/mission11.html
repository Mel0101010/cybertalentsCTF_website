Mission #11: Analyse de Logs SOC

STATUT: EN COURS

📋 Briefing de Mission

Dans cette mission, vous devez créer un programme qui analyse des logs de sécurité (SOC - Security Operations Center). Le serveur vous enverra des logs à analyser pour détecter des activités suspectes et des tentatives d'intrusion.

Votre programme devra identifier les patterns d'attaque, les anomalies et générer des alertes appropriées.

📡 Informations du Serveur

IP : ?.?.?.?

Port : 9011

Protocole : TCP

📝 Types de Logs SOC

Les logs SOC contiennent différents types d'informations :

  • Logs d'authentification : tentatives de connexion, succès, échecs
  • Logs réseau : connexions, ports, protocoles
  • Logs système : accès aux fichiers, modifications, exécutions
  • Logs d'application : requêtes, erreurs, comportements anormaux
  • Logs de sécurité : alertes, violations, tentatives d'intrusion

💻 Exemple de Code Python

import socket
import re
from datetime import datetime, timedelta
from collections import defaultdict

class LogAnalyzer:
    def __init__(self):
        self.auth_attempts = defaultdict(list)
        self.port_connections = defaultdict(list)
        self.file_transfers = defaultdict(list)
        self.alert_thresholds = {
            'brute_force': {'count': 5, 'window': 300},  # 5 minutes
            'port_scan': {'count': 4, 'window': 60},     # 1 minute
            'ddos': {'count': 1000, 'window': 60},       # 1 minute
            'data_exfil': {'size': 100, 'window': 300}   # 100MB in 5 minutes
        }

    def parse_log_line(self, line):
        """Parse une ligne de log et extrait les informations importantes."""
        pattern = r'\[(.*?)\] \[(.*?)\] \[(.*?)\] (.*)'
        match = re.match(pattern, line)
        if match:
            timestamp, log_type, status, details = match.groups()
            return {
                'timestamp': datetime.strptime(timestamp, '%Y-%m-%d %H:%M:%S'),
                'type': log_type,
                'status': status,
                'details': details
            }
        return None

    def check_brute_force(self, log_entry):
        """Vérifie les tentatives de brute force."""
        if log_entry['type'] == 'AUTH' and log_entry['status'] == 'FAIL':
            ip = re.search(r'IP:(\S+)', log_entry['details']).group(1)
            user = re.search(r'User:(\S+)', log_entry['details']).group(1)
            
            # Ajouter la tentative
            self.auth_attempts[(ip, user)].append(log_entry['timestamp'])
            
            # Nettoyer les anciennes tentatives
            window = timedelta(seconds=self.alert_thresholds['brute_force']['window'])
            self.auth_attempts[(ip, user)] = [
                t for t in self.auth_attempts[(ip, user)]
                if log_entry['timestamp'] - t <= window
            ]
            
            # Vérifier le seuil
            if len(self.auth_attempts[(ip, user)]) >= self.alert_thresholds['brute_force']['count']:
                return {
                    'type': 'BRUTE_FORCE',
                    'ip': ip,
                    'user': user,
                    'attempts': len(self.auth_attempts[(ip, user)]),
                    'time_start': min(self.auth_attempts[(ip, user)]),
                    'time_end': max(self.auth_attempts[(ip, user)]),
                    'level': 'HIGH'
                }
        return None

    def check_port_scan(self, log_entry):
        """Vérifie les scans de port."""
        if log_entry['type'] == 'NET' and log_entry['status'] == 'CONN':
            ip = re.search(r'IP:(\S+)', log_entry['details']).group(1)
            port = re.search(r'Port:(\d+)', log_entry['details']).group(1)
            
            # Ajouter la connexion
            self.port_connections[(ip, port)].append(log_entry['timestamp'])
            
            # Nettoyer les anciennes connexions
            window = timedelta(seconds=self.alert_thresholds['port_scan']['window'])
            self.port_connections[(ip, port)] = [
                t for t in self.port_connections[(ip, port)]
                if log_entry['timestamp'] - t <= window
            ]
            
            # Vérifier le seuil
            if len(self.port_connections[(ip, port)]) >= self.alert_thresholds['port_scan']['count']:
                return {
                    'type': 'PORT_SCAN',
                    'ip': ip,
                    'port': port,
                    'attempts': len(self.port_connections[(ip, port)]),
                    'time_start': min(self.port_connections[(ip, port)]),
                    'time_end': max(self.port_connections[(ip, port)]),
                    'level': 'MEDIUM'
                }
        return None

    def analyze_logs(self, logs):
        """Analyse un ensemble de logs et génère des alertes."""
        alerts = []
        for line in logs.split('\n'):
            if not line.strip():
                continue
            
            log_entry = self.parse_log_line(line)
            if not log_entry:
                continue
            
            # Vérifier différents types d'attaques
            brute_force_alert = self.check_brute_force(log_entry)
            if brute_force_alert:
                alerts.append(brute_force_alert)
            
            port_scan_alert = self.check_port_scan(log_entry)
            if port_scan_alert:
                alerts.append(port_scan_alert)
        
        return alerts

    def format_alert(self, alert):
        """Formate une alerte pour l'affichage."""
        return f"""ALERT: {alert['type']}
IP: {alert['ip']}
{'User: ' + alert['user'] if 'user' in alert else 'Port: ' + alert['port']}
Attempts: {alert['attempts']}
Time: {alert['time_start'].strftime('%Y-%m-%d %H:%M:%S')}-{alert['time_end'].strftime('%H:%M:%S')}
Level: {alert['level']}"""

# Créer un socket TCP
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

# Se connecter au serveur
s.connect(('IP_DU_SERVEUR', 9011))

# Créer l'analyseur de logs
analyzer = LogAnalyzer()

while True:
    # Recevoir les logs
    logs = s.recv(4096).decode().strip()
    if not logs:
        break
    
    # Analyser les logs
    alerts = analyzer.analyze_logs(logs)
    
    if alerts:
        # Envoyer la première alerte
        response = analyzer.format_alert(alerts[0])
        s.send(response.encode())
        
        # Recevoir la confirmation
        confirmation = s.recv(1024).decode()
        print(confirmation)
        
        if "flag" in confirmation.lower():
            break
    else:
        s.send(b"No alerts generated")

# Fermer la connexion
s.close()
