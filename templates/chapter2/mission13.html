Mission #13: Analyse de Malware

STATUT: EN COURS

📋 Briefing de Mission

Dans cette mission, vous devez créer un programme qui analyse des échantillons de malware. Le serveur vous enverra des fichiers suspects à analyser pour identifier leur type, leur comportement et leurs caractéristiques.

Votre programme devra extraire les informations importantes, détecter les patterns malveillants et générer un rapport d'analyse.

📡 Informations du Serveur

IP : ?.?.?.?

Port : 9013

Protocole : TCP

📝 Types de Malware

Les échantillons peuvent appartenir à différentes catégories :

  • Virus : Se réplique en infectant d'autres fichiers
  • Vers : Se propage via le réseau
  • Trojans : Se fait passer pour un logiciel légitime
  • Ransomware : Chiffre les fichiers et demande une rançon
  • Backdoors : Permet un accès distant non autorisé

💻 Exemple de Code Python

import socket
import re
import hashlib
from datetime import datetime

class MalwareAnalyzer:
    def __init__(self):
        self.malware_patterns = {
            'backdoor': [
                r'backdoor\.exe',
                r'net user /add',
                r'net localgroup administrators',
                r'powershell\.exe -enc',
                r'http://.*/payload'
            ],
            'trojan': [
                r'VBA',
                r'AutoOpen',
                r'Shell',
                r'certutil\.exe',
                r'http://.*/macro'
            ],
            'ransomware': [
                r'encrypt',
                r'decrypt',
                r'ransom',
                r'bitcoin',
                r'\.encrypted'
            ],
            'worm': [
                r'exploit',
                r'vulnerability',
                r'scan',
                r'port',
                r'network'
            ]
        }
        
        self.suspicious_commands = {
            'system': [
                'cmd.exe', 'powershell.exe', 'certutil.exe',
                'net.exe', 'reg.exe', 'schtasks.exe'
            ],
            'network': [
                'download', 'upload', 'connect', 'socket',
                'http://', 'https://', 'ftp://'
            ],
            'persistence': [
                'HKCU\\', 'HKLM\\', 'Startup', 'Task',
                'Service', 'Scheduled'
            ]
        }

    def calculate_hashes(self, content):
        """Calcule les hashes MD5, SHA1 et SHA256 du contenu."""
        return {
            'MD5': hashlib.md5(content).hexdigest(),
            'SHA1': hashlib.sha1(content).hexdigest(),
            'SHA256': hashlib.sha256(content).hexdigest()
        }

    def extract_strings(self, content):
        """Extrait les chaînes de caractères du contenu."""
        # Convertir le contenu en texte
        try:
            text = content.decode('utf-8', errors='ignore')
        except:
            text = str(content)
        
        # Extraire les chaînes non-ASCII
        strings = re.findall(r'[ -~]{4,}', text)
        return [s for s in strings if any(c.isprintable() for c in s)]

    def analyze_behavior(self, strings):
        """Analyse le comportement basé sur les chaînes extraites."""
        behavior = {
            'system_commands': [],
            'network_activity': [],
            'persistence': [],
            'suspicious_patterns': []
        }
        
        for s in strings:
            # Vérifier les commandes système
            for cmd in self.suspicious_commands['system']:
                if cmd.lower() in s.lower():
                    behavior['system_commands'].append(s)
            
            # Vérifier l'activité réseau
            for net in self.suspicious_commands['network']:
                if net.lower() in s.lower():
                    behavior['network_activity'].append(s)
            
            # Vérifier la persistance
            for pers in self.suspicious_commands['persistence']:
                if pers.lower() in s.lower():
                    behavior['persistence'].append(s)
            
            # Vérifier les patterns malveillants
            for malware_type, patterns in self.malware_patterns.items():
                for pattern in patterns:
                    if re.search(pattern, s, re.IGNORECASE):
                        behavior['suspicious_patterns'].append((malware_type, s))
        
        return behavior

    def determine_type(self, behavior):
        """Détermine le type de malware basé sur le comportement."""
        type_scores = {
            'backdoor': 0,
            'trojan': 0,
            'ransomware': 0,
            'worm': 0
        }
        
        # Analyser les patterns suspects
        for malware_type, _ in behavior['suspicious_patterns']:
            type_scores[malware_type] += 1
        
        # Analyser les commandes système
        if behavior['system_commands']:
            type_scores['backdoor'] += 1
            type_scores['trojan'] += 1
        
        # Analyser l'activité réseau
        if behavior['network_activity']:
            type_scores['worm'] += 1
            type_scores['backdoor'] += 1
        
        # Analyser la persistance
        if behavior['persistence']:
            type_scores['trojan'] += 1
            type_scores['backdoor'] += 1
        
        # Déterminer le type avec le score le plus élevé
        max_score = max(type_scores.values())
        if max_score == 0:
            return 'Unknown'
        
        return max(type_scores.items(), key=lambda x: x[1])[0]

    def analyze_sample(self, sample_data):
        """Analyse un échantillon de malware et génère un rapport."""
        # Extraire les informations de base
        name_match = re.search(r'Name: (.*)', sample_data)
        size_match = re.search(r'Size: (\d+) bytes', sample_data)
        type_match = re.search(r'Type: (.*)', sample_data)
        
        if not all([name_match, size_match, type_match]):
            return None
        
        name = name_match.group(1)
        size = int(size_match.group(1))
        file_type = type_match.group(1)
        
        # Extraire les hashes
        md5_match = re.search(r'MD5: (.*)', sample_data)
        sha1_match = re.search(r'SHA1: (.*)', sample_data)
        sha256_match = re.search(r'SHA256: (.*)', sample_data)
        
        hashes = {
            'MD5': md5_match.group(1) if md5_match else None,
            'SHA1': sha1_match.group(1) if sha1_match else None,
            'SHA256': sha256_match.group(1) if sha256_match else None
        }
        
        # Extraire les chaînes
        strings_section = re.search(r'\[STRINGS\](.*?)(?=\[|$)', sample_data, re.DOTALL)
        if strings_section:
            strings = [s.strip() for s in strings_section.group(1).split('\n') if s.strip()]
        else:
            strings = []
        
        # Analyser le comportement
        behavior = self.analyze_behavior(strings)
        malware_type = self.determine_type(behavior)
        
        # Générer les indicateurs
        indicators = []
        if behavior['system_commands']:
            indicators.append('- Executes system commands')
        if behavior['network_activity']:
            indicators.append('- Performs network activity')
        if behavior['persistence']:
            indicators.append('- Attempts persistence')
        if behavior['suspicious_patterns']:
            indicators.append('- Contains malicious patterns')
        
        # Déterminer le niveau de confiance
        confidence = 'HIGH' if len(indicators) >= 3 else 'MEDIUM' if len(indicators) >= 2 else 'LOW'
        
        return {
            'type': malware_type.upper(),
            'name': name,
            'size': size,
            'hashes': hashes,
            'indicators': indicators,
            'confidence': confidence
        }

    def format_analysis(self, analysis):
        """Formate l'analyse pour l'affichage."""
        return f"""ANALYSIS: {analysis['type']}
Type: {analysis['type']}
Name: {analysis['name']}
Size: {analysis['size']} bytes
Hashes:
  MD5: {analysis['hashes']['MD5']}
  SHA1: {analysis['hashes']['SHA1']}
  SHA256: {analysis['hashes']['SHA256']}
Indicators:
{chr(10).join('  ' + i for i in analysis['indicators'])}
Confidence: {analysis['confidence']}"""

# Créer un socket TCP
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

# Se connecter au serveur
s.connect(('IP_DU_SERVEUR', 9013))

# Créer l'analyseur de malware
analyzer = MalwareAnalyzer()

while True:
    # Recevoir l'échantillon
    sample = s.recv(4096).decode().strip()
    if not sample:
        break
    
    # Analyser l'échantillon
    analysis = analyzer.analyze_sample(sample)
    
    if analysis:
        # Envoyer l'analyse
        response = analyzer.format_analysis(analysis)
        s.send(response.encode())
        
        # Recevoir la confirmation
        confirmation = s.recv(1024).decode()
        print(confirmation)
        
        if "flag" in confirmation.lower():
            break
    else:
        s.send(b"Invalid sample format")

# Fermer la connexion
s.close()
