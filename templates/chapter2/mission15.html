Mission #15: Analyse de Cryptographie

STATUT: EN COURS

📋 Briefing de Mission

Dans cette mission, vous devez créer un programme qui analyse et déchiffre des messages cryptés. Le serveur vous enverra des messages chiffrés à analyser pour déterminer leur méthode de chiffrement, leur clé et leur contenu.

Votre programme devra identifier le type de chiffrement, extraire les paramètres importants et déchiffrer le message pour obtenir le flag.

📡 Informations du Serveur

IP : ?.?.?.?

Port : 9015

Protocole : TCP

📝 Types de Chiffrement

Les messages peuvent être chiffrés avec différentes méthodes :

  • Chiffrement de César : Décalage simple des lettres
  • Chiffrement de Vigenère : Chiffrement polyalphabétique
  • RSA : Chiffrement asymétrique
  • AES : Chiffrement symétrique moderne
  • Base64 : Encodage (non chiffrement)

💻 Exemple de Code Python

import socket
import re
import base64
from datetime import datetime
from Crypto.PublicKey import RSA
from Crypto.Cipher import PKCS1_OAEP, AES
from Crypto.Util.Padding import unpad

class CryptoAnalyzer:
    def __init__(self):
        self.cipher_patterns = {
            'caesar': [
                (r'Type: Caesar', 'header'),
                (r'Key: (\d+)', 'key'),
                (r'[A-Za-z]', 'text')
            ],
            'vigenere': [
                (r'Type: Vigenere', 'header'),
                (r'Key: ([A-Za-z]+)', 'key'),
                (r'[A-Za-z]', 'text')
            ],
            'rsa': [
                (r'Type: RSA', 'header'),
                (r'Key: (-----BEGIN.*?-----.*?-----END.*?-----)', 'key', re.DOTALL),
                (r'[A-Za-z0-9+/=]{32,}', 'base64')
            ],
            'aes': [
                (r'Type: AES', 'header'),
                (r'Key: ([A-Za-z0-9+/=]{32,})', 'key'),
                (r'IV: ([A-Za-z0-9+/=]{24,})', 'iv'),
                (r'[A-Za-z0-9+/=]{32,}', 'base64')
            ],
            'base64': [
                (r'Type: Base64', 'header'),
                (r'[A-Za-z0-9+/=]{32,}', 'base64')
            ]
        }
        
        self.vulnerability_patterns = {
            'caesar': [
                (r'[Ee]', 'high_frequency'),
                (r'[QqXxZz]', 'low_frequency'),
                (r'[^A-Za-z\s]', 'special_chars')
            ],
            'vigenere': [
                (r'([A-Za-z])\1{2,}', 'repeated_chars'),
                (r'[Ee]{2,}', 'common_pattern'),
                (r'[^A-Za-z\s]', 'special_chars')
            ],
            'rsa': [
                (r'-----BEGIN.*?PRIVATE.*?-----', 'private_key_exposed'),
                (r'[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}', 'ip_address'),
                (r'[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}', 'email')
            ],
            'aes': [
                (r'[A-Za-z0-9+/=]{16}', 'block_size'),
                (r'[A-Za-z0-9+/=]{32,}', 'multiple_blocks'),
                (r'[^A-Za-z0-9+/=\s]', 'invalid_chars')
            ],
            'base64': [
                (r'[^A-Za-z0-9+/=\s]', 'invalid_chars'),
                (r'[A-Za-z0-9+/=]{1,}(?![A-Za-z0-9+/=])', 'incomplete_block'),
                (r'[A-Za-z0-9+/=]{4,}(?![A-Za-z0-9+/=]{4})', 'padding_error')
            ]
        }

    def extract_message_info(self, message_data):
        """Extrait les informations de base du message."""
        info = {}
        
        # Extraire les informations de base
        type_match = re.search(r'Type: (.*)', message_data)
        key_match = re.search(r'Key: (.*)', message_data)
        format_match = re.search(r'Format: (.*)', message_data)
        length_match = re.search(r'Length: (\d+) bytes', message_data)
        
        if all([type_match, format_match, length_match]):
            info['type'] = type_match.group(1).lower()
            info['format'] = format_match.group(1)
            info['length'] = int(length_match.group(1))
            
            if key_match:
                info['key'] = key_match.group(1)
        
        # Extraire le timestamp
        timestamp_match = re.search(r'Timestamp: (\d+)', message_data)
        if timestamp_match:
            info['timestamp'] = int(timestamp_match.group(1))
        
        return info

    def extract_headers(self, message_data):
        """Extrait les en-têtes du message."""
        headers_section = re.search(r'\[HEADERS\](.*?)(?=\[|$)', message_data, re.DOTALL)
        if not headers_section:
            return []
        
        headers = []
        for line in headers_section.group(1).split('\n'):
            line = line.strip()
            if line:
                headers.append(line)
        
        return headers

    def extract_payload(self, message_data):
        """Extrait le payload du message."""
        payload_section = re.search(r'\[PAYLOAD\](.*?)(?=\[|$)', message_data, re.DOTALL)
        if not payload_section:
            return None
        
        return payload_section.group(1).strip()

    def identify_cipher(self, headers, payload):
        """Identifie le type de chiffrement basé sur les en-têtes et le payload."""
        cipher_scores = {
            'caesar': 0,
            'vigenere': 0,
            'rsa': 0,
            'aes': 0,
            'base64': 0
        }
        
        # Analyser les en-têtes
        for header in headers:
            for cipher, patterns in self.cipher_patterns.items():
                for pattern, _ in patterns:
                    if re.search(pattern, header, re.IGNORECASE):
                        cipher_scores[cipher] += 1
        
        # Analyser le payload si présent
        if payload:
            for cipher, patterns in self.cipher_patterns.items():
                for pattern, _ in patterns:
                    if re.search(pattern, payload, re.IGNORECASE):
                        cipher_scores[cipher] += 1
        
        # Déterminer le chiffrement avec le score le plus élevé
        max_score = max(cipher_scores.values())
        if max_score == 0:
            return 'Unknown'
        
        return max(cipher_scores.items(), key=lambda x: x[1])[0]

    def analyze_vulnerabilities(self, cipher, headers, payload):
        """Analyse les vulnérabilités dans le message."""
        vulnerabilities = []
        
        if cipher in self.vulnerability_patterns:
            # Vérifier les en-têtes
            for header in headers:
                for pattern, vuln_type in self.vulnerability_patterns[cipher]:
                    if re.search(pattern, header, re.IGNORECASE):
                        vulnerabilities.append(vuln_type)
            
            # Vérifier le payload si présent
            if payload:
                for pattern, vuln_type in self.vulnerability_patterns[cipher]:
                    if re.search(pattern, payload, re.IGNORECASE):
                        vulnerabilities.append(vuln_type)
        
        return vulnerabilities

    def decrypt_caesar(self, text, key):
        """Déchiffre un texte chiffré avec le chiffrement de César."""
        result = ""
        for char in text:
            if char.isalpha():
                ascii_offset = ord('A') if char.isupper() else ord('a')
                result += chr((ord(char) - ascii_offset - int(key)) % 26 + ascii_offset)
            else:
                result += char
        return result

    def decrypt_vigenere(self, text, key):
        """Déchiffre un texte chiffré avec le chiffrement de Vigenère."""
        result = ""
        key_length = len(key)
        key_as_int = [ord(k.upper()) - ord('A') for k in key]
        text_int = [ord(t.upper()) - ord('A') if t.isalpha() else t for t in text]
        
        for i, char in enumerate(text_int):
            if isinstance(char, int):
                result += chr((char - key_as_int[i % key_length]) % 26 + ord('A'))
            else:
                result += char
        
        return result

    def decrypt_rsa(self, text, key_data):
        """Déchiffre un texte chiffré avec RSA."""
        try:
            # Convertir la clé en format RSA
            key = RSA.import_key(key_data)
            cipher = PKCS1_OAEP.new(key)
            
            # Déchiffrer le texte en base64
            encrypted_data = base64.b64decode(text)
            decrypted_data = cipher.decrypt(encrypted_data)
            
            return decrypted_data.decode()
        except Exception as e:
            return f"Erreur de déchiffrement RSA: {str(e)}"

    def decrypt_aes(self, text, key, iv):
        """Déchiffre un texte chiffré avec AES."""
        try:
            # Convertir la clé et l'IV
            key_data = base64.b64decode(key)
            iv_data = base64.b64decode(iv)
            
            # Créer le déchiffreur AES
            cipher = AES.new(key_data, AES.MODE_CBC, iv_data)
            
            # Déchiffrer le texte en base64
            encrypted_data = base64.b64decode(text)
            decrypted_data = unpad(cipher.decrypt(encrypted_data), AES.block_size)
            
            return decrypted_data.decode()
        except Exception as e:
            return f"Erreur de déchiffrement AES: {str(e)}"

    def decode_base64(self, text):
        """Décode un texte encodé en base64."""
        try:
            return base64.b64decode(text).decode()
        except Exception as e:
            return f"Erreur de décodage base64: {str(e)}"

    def analyze_message(self, message_data):
        """Analyse un message chiffré et génère un rapport."""
        # Extraire les informations de base
        message_info = self.extract_message_info(message_data)
        if not message_info:
            return None
        
        # Extraire les en-têtes et le payload
        headers = self.extract_headers(message_data)
        payload = self.extract_payload(message_data)
        
        # Identifier le chiffrement
        cipher = self.identify_cipher(headers, payload)
        
        # Analyser les vulnérabilités
        vulnerabilities = self.analyze_vulnerabilities(cipher, headers, payload)
        
        # Déchiffrer le message
        decrypted = None
        if cipher == 'caesar' and 'key' in message_info:
            decrypted = self.decrypt_caesar(payload, message_info['key'])
        elif cipher == 'vigenere' and 'key' in message_info:
            decrypted = self.decrypt_vigenere(payload, message_info['key'])
        elif cipher == 'rsa' and 'key' in message_info:
            decrypted = self.decrypt_rsa(payload, message_info['key'])
        elif cipher == 'aes' and 'key' in message_info:
            iv_match = re.search(r'IV: ([A-Za-z0-9+/=]{24,})', message_data)
            if iv_match:
                decrypted = self.decrypt_aes(payload, message_info['key'], iv_match.group(1))
        elif cipher == 'base64':
            decrypted = self.decode_base64(payload)
        
        # Déterminer le niveau de confiance
        confidence = 'HIGH' if decrypted and not vulnerabilities else 'MEDIUM' if decrypted else 'LOW'
        
        return {
            'type': cipher.upper(),
            'info': message_info,
            'decrypted': decrypted,
            'vulnerabilities': vulnerabilities,
            'confidence': confidence
        }

    def format_analysis(self, analysis):
        """Formate l'analyse pour l'affichage."""
        output = [f"ANALYSIS: {analysis['type']}"]
        
        # Ajouter les détails spécifiques au chiffrement
        if analysis['type'] == 'CAESAR':
            output.append("Type: Caesar Cipher")
            output.append(f"Key: {analysis['info']['key']}")
            output.append("Method: Shift")
        elif analysis['type'] == 'VIGENERE':
            output.append("Type: Vigenere Cipher")
            output.append(f"Key: {analysis['info']['key']}")
            output.append("Method: Polyalphabetic")
        elif analysis['type'] == 'RSA':
            output.append("Type: RSA Cipher")
            output.append("Method: Asymmetric")
        elif analysis['type'] == 'AES':
            output.append("Type: AES Cipher")
            output.append("Method: Symmetric Block")
        elif analysis['type'] == 'BASE64':
            output.append("Type: Base64 Encoding")
            output.append("Method: Encoding")
        
        # Ajouter le message déchiffré
        if analysis['decrypted']:
            output.append(f"Original: {analysis['decrypted']}")
        
        # Ajouter les vulnérabilités si présentes
        if analysis['vulnerabilities']:
            output.append("Vulnerabilities:")
            for vuln in analysis['vulnerabilities']:
                output.append(f"  - {vuln}")
        
        # Ajouter le niveau de confiance
        output.append(f"Confidence: {analysis['confidence']}")
        
        return '\n'.join(output)

# Créer un socket TCP
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

# Se connecter au serveur
s.connect(('IP_DU_SERVEUR', 9015))

# Créer l'analyseur de cryptographie
analyzer = CryptoAnalyzer()

while True:
    # Recevoir le message
    message = s.recv(4096).decode().strip()
    if not message:
        break
    
    # Analyser le message
    analysis = analyzer.analyze_message(message)
    
    if analysis:
        # Envoyer l'analyse
        response = analyzer.format_analysis(analysis)
        s.send(response.encode())
        
        # Recevoir la confirmation
        confirmation = s.recv(1024).decode()
        print(confirmation)
        
        if "flag" in confirmation.lower():
            break
    else:
        s.send(b"Invalid message format")

# Fermer la connexion
s.close()
